Authentication Flow sa Frontend

Matutunan kung paano i-handle ang authentication sa Next.js — mula sa pag-store ng tokens hanggang sa pag-protect ng routes gamit ang middleware.

Lesson Goal Sa lesson na ito, magagawa mong mag-implement ng complete login/logout flow at i-protect ang dashboard pages gamit ang Next.js middleware.

Storing Auth Tokens

Kapag nag-login ang user sa Laravel API, nagre-return ito ng token (JWT o Sanctum token). Ang tanong: saan ito i-store?

Option 1: localStorage (Client-Side) — Ito ang pinaka-simple. I-store ang token sa localStorage at i-attach sa bawat request headers.

// lib/auth.js
export const auth = {
  getToken() {
    if (typeof window !== 'undefined') {
      return localStorage.getItem('token');
    }
    return null;
  },

  setToken(token) {
    localStorage.setItem('token', token);
  },

  removeToken() {
    localStorage.removeItem('token');
  },

  isAuthenticated() {
    return !!this.getToken();
  }
};

// Usage sa API calls
async function fetchWithAuth(endpoint, options = {}) {
  const token = auth.getToken();

  const res = await fetch(`http://localhost:8000/api${endpoint}`, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      'Accept': 'application/json',
      'Authorization': token ? `Bearer ${token}` : '',
      ...options.headers,
    },
  });

  if (res.status === 401) {
    auth.removeToken();
    window.location.href = '/login';
    return;
  }

  return res.json();
}

Option 2: HttpOnly Cookies (More Secure) — Kapag gumagamit ng Laravel Sanctum sa SPA mode, ang token ay naka-store sa httpOnly cookie. Hindi mo ito kailangan i-store sa localStorage — awtomatikong isesend ng browser ang cookie sa bawat request.

Para sa ating Laravel + Next.js setup, ang localStorage approach ay sapat na habang nagde-develop. Pag production, pwede mong i-switch sa httpOnly cookies para sa mas mataas na security.

Login at Logout Flow

Ito ang complete login flow — mula sa form hanggang sa Laravel API:

'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { auth } from '@/lib/auth';

export default function LoginForm() {
  const router = useRouter();
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);

  async function handleSubmit(e) {
    e.preventDefault();
    setLoading(true);
    setError('');

    try {
      const res = await fetch('http://localhost:8000/api/login', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json',
        },
        body: JSON.stringify({ email, password }),
      });

      const data = await res.json();

      if (!res.ok) {
        setError(data.message || 'Invalid credentials');
        return;
      }

      // I-store ang token
      auth.setToken(data.token);

      // I-store ang user info (optional pero useful)
      localStorage.setItem('user', JSON.stringify(data.user));

      // Redirect sa dashboard
      router.push('/dashboard');
      router.refresh();
    } catch (err) {
      setError('Network error. Please try again.');
    } finally {
      setLoading(false);
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      {error && <div className="alert alert-error">{error}</div>}

      <div className="form-group">
        <label htmlFor="email">Email</label>
        <input id="email" type="email" value={email}
               onChange={(e) => setEmail(e.target.value)} required />
      </div>

      <div className="form-group">
        <label htmlFor="password">Password</label>
        <input id="password" type="password" value={password}
               onChange={(e) => setPassword(e.target.value)} required />
      </div>

      <button type="submit" disabled={loading}>
        {loading ? 'Logging in...' : 'Login'}
      </button>
    </form>
  );
}

Para naman sa logout:

// components/layout/LogoutButton.js
'use client';

import { useRouter } from 'next/navigation';
import { auth } from '@/lib/auth';

export default function LogoutButton() {
  const router = useRouter();

  async function handleLogout() {
    try {
      await fetch('http://localhost:8000/api/logout', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${auth.getToken()}`,
          'Accept': 'application/json',
        },
      });
    } catch (err) {
      // Kahit mag-fail ang API, i-clear pa rin ang local token
    }

    auth.removeToken();
    localStorage.removeItem('user');
    router.push('/login');
    router.refresh();
  }

  return (
    <button onClick={handleLogout}>Logout</button>
  );
}

Auth Context

Para hindi mo kailangan i-pass ang user info sa bawat component, gumamit ng React Context. Ito ang magse-set ng "sinong user ang naka-login" state:

'use client';

import { createContext, useContext, useState, useEffect } from 'react';
import { auth } from '@/lib/auth';

const AuthContext = createContext(null);

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Check kung may existing token
    const token = auth.getToken();
    const storedUser = localStorage.getItem('user');

    if (token && storedUser) {
      try {
        setUser(JSON.parse(storedUser));
      } catch {
        auth.removeToken();
        localStorage.removeItem('user');
      }
    }
    setLoading(false);
  }, []);

  function login(userData, token) {
    auth.setToken(token);
    localStorage.setItem('user', JSON.stringify(userData));
    setUser(userData);
  }

  function logout() {
    auth.removeToken();
    localStorage.removeItem('user');
    setUser(null);
  }

  return (
    <AuthContext.Provider value={{ user, loading, login, logout }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error('useAuth must be used within AuthProvider');
  }
  return context;
}

I-wrap ang entire app sa AuthProvider sa root layout:

// app/layout.js
import { AuthProvider } from '@/lib/auth-context';

export default function RootLayout({ children }) {
  return (
    <html lang="en">
      <body>
        <AuthProvider>
          {children}
        </AuthProvider>
      </body>
    </html>
  );
}

Protected Routes gamit ang Middleware

Ang Next.js middleware ay nagre-run sa server bago ang page render — perpekto ito para i-check ang auth status. Gumawa ng middleware.js sa root ng project:

// middleware.js
import { NextResponse } from 'next/server';

export function middleware(request) {
  const token = request.cookies.get('token')?.value;
  const { pathname } = request.nextUrl;

  // Public routes — walang auth required
  const publicPaths = ['/login', '/register', '/'];
  const isPublicPath = publicPaths.some(path => pathname === path);

  // Kung hindi authenticated at hindi public path — redirect sa login
  if (!token && !isPublicPath) {
    return NextResponse.redirect(new URL('/login', request.url));
  }

  // Kung authenticated at nasa login/register — redirect sa dashboard
  if (token && (pathname === '/login' || pathname === '/register')) {
    return NextResponse.redirect(new URL('/dashboard', request.url));
  }

  return NextResponse.next();
}

export const config = {
  matcher: [
    '/dashboard/:path*',
    '/listings/new',
    '/profile/:path*',
    '/login',
    '/register',
  ],
};

Important: Ang middleware sa Next.js ay nagre-run sa server, kaya hindi mo maa-access ang localStorage. Sa halip, gumamit ng cookies. Kapag nag-login ka, i-set mo rin ang token bilang cookie:

// Pagkatapos ng login, i-set ang cookie
document.cookie = `token=${data.token}; path=/; max-age=86400; SameSite=Lax`;

Ito ang mas secure na approach — ang token ay naka-cookie at hindi accessible sa JavaScript, pero nasesend ng browser sa bawat request automatically.

Quiz

Summary

  • I-store ang auth token sa localStorage (simple) o httpOnly cookie (mas secure). Sa middleware, gumamit ng cookies dahil hindi accessible ang localStorage.
  • Ang login flow ay: POST sa Laravel API, i-store ang token, i-redirect sa dashboard. Ang logout flow ay: POST sa API, i-clear ang token, i-redirect sa login.
  • Gumamit ng AuthContext para i-manage ang user state sa buong app — hindi mo kailangan i-pass ang user sa bawat component.
  • Ang Next.js middleware ay perpekto para sa route protection — nagre-run sa server bago ang page render.
