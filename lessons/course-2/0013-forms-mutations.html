Forms & Mutations

Matutunan kung paano mag-handle ng forms sa Next.js at mag-send ng POST, PUT, DELETE requests sa Laravel API.

Lesson Goal Sa lesson na ito, magagawa mong gumawa ng create/edit form para sa core entity na may validation at error display mula sa Laravel.

Client-Side Forms sa Next.js

Ang forms ay kailangang maging client components dahil nangangailangan sila ng useState, onChange handlers, at event handling. Ang pattern ay simple: i-mark ang form component bilang 'use client', i-manage ang form state, at i-submit sa Laravel API.

'use client';

import { useState } from 'react';

export default function ListingForm({ initialData, onSubmit }) {
  const [formData, setFormData] = useState({
    title: initialData?.title || '',
    description: initialData?.description || '',
    price: initialData?.price || '',
  });
  const [errors, setErrors] = useState({});
  const [loading, setLoading] = useState(false);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData(prev => ({ ...prev, [name]: value }));
    // Clear error when user types
    if (errors[name]) {
      setErrors(prev => ({ ...prev, [name]: '' }));
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      <div>
        <label htmlFor="title">Title</label>
        <input
          id="title"
          name="title"
          value={formData.title}
          onChange={handleChange}
        />
        {errors.title && <p className="error">{errors.title}</p>}
      </div>
      {/* ... more fields ... */}
    </form>
  );
}

Ang key pattern dito ay: (1) i-manage ang form state gamit ang useState, (2) i-handle ang onChange para i-update ang state, at (3) i-display ang errors mula sa Laravel validation.

Sending Data sa Laravel API

Para sa mutations (create, update, delete), gumamit ng fetch sa client side. Ito ang complete pattern para sa form submission:

'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';

export default function CreateListingForm() {
  const router = useRouter();
  const [formData, setFormData] = useState({
    title: '',
    description: '',
    price: '',
  });
  const [errors, setErrors] = useState({});
  const [loading, setLoading] = useState(false);

  async function handleSubmit(e) {
    e.preventDefault();
    setLoading(true);
    setErrors({});

    try {
      const res = await fetch('http://localhost:8000/api/listings', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('token')}`,
        },
        body: JSON.stringify(formData),
      });

      const data = await res.json();

      if (!res.ok) {
        // Laravel validation errors
        if (res.status === 422 && data.errors) {
          setErrors(data.errors);
          return;
        }
        throw new Error(data.message || 'Something went wrong');
      }

      // Success — redirect sa listing page
      router.push(`/listings/${data.id}`);
      router.refresh(); // I-refresh ang server components

    } catch (error) {
      setErrors({ general: error.message });
    } finally {
      setLoading(false);
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      {errors.general && (
        <div className="alert alert-error">{errors.general}</div>
      )}
      {/* form fields */}
      <button type="submit" disabled={loading}>
        {loading ? 'Saving...' : 'Create Listing'}
      </button>
    </form>
  );
}

Ang router.refresh() ay mahalaga — ito ang nag-t-trigger ng re-render ng server components para makita agad ang bagong data. Hindi mo na kailangan ng manual state update para sa list.

Handling Laravel Validation Errors

Ang Laravel ay nagre-return ng 422 status code kapag may validation errors, na may format na ganito:

{
  "message": "The given data was invalid.",
  "errors": {
    "title": ["The title field is required."],
    "price": ["The price must be a number."]
  }
}

I-display ang mga errors na ito sa form. Ang pattern ay i-map ang error keys sa corresponding fields:

function FieldError({ errors, field }) {
  if (!errors[field]) return null;
  return (
    <ul className="error-list">
      {errors[field].map((error, i) => (
        <li key={i} className="error">{error}</li>
      ))}
    </ul>
  );
}

// Sa form:
<div>
  <label htmlFor="title">Title</label>
  <input id="title" name="title" value={formData.title} onChange={handleChange} />
  <FieldError errors={errors} field="title" />
</div>

Ito ang cleanest na approach — reusable na field error component na nagre-render ng validation messages mula sa Laravel.

Update at Delete Mutations

Para sa update at delete, ang pattern ay halos kapareho — iba lang ang HTTP method at URL:

// Update — PUT request
async function handleUpdate(id, formData) {
  const res = await fetch(`http://localhost:8000/api/listings/${id}`, {
    method: 'PUT',
    headers: {
      'Content-Type': 'application/json',
      'Accept': 'application/json',
      'Authorization': `Bearer ${localStorage.getItem('token')}`,
    },
    body: JSON.stringify(formData),
  });

  if (!res.ok) {
    const data = await res.json();
    throw new Error(data.message || 'Update failed');
  }

  return res.json();
}

// Delete — DELETE request
async function handleDelete(id) {
  const res = await fetch(`http://localhost:8000/api/listings/${id}`, {
    method: 'DELETE',
    headers: {
      'Accept': 'application/json',
      'Authorization': `Bearer ${localStorage.getItem('token')}`,
    },
  });

  if (!res.ok) {
    throw new Error('Delete failed');
  }
}

Para sa delete, laging mag-confirm muna bago mag-send ng request. Pwede mong gamitin ang window.confirm() o isang custom modal component.

Practical: Complete Create Listing Flow

Ito ang buong flow — mula sa form hanggang sa Laravel API. Ito ang gagamitin mo sa actual project:

// app/listings/new/page.js
import CreateListingForm from '@/components/listings/CreateListingForm';

export default function NewListingPage() {

    <div>
      <h1>Create New Listing</h1>
      <CreateListingForm />

// components/listings/CreateListingForm.js
'use client';

import { useState } from 'react';
import { useRouter } from 'next/navigation';

export default function CreateListingForm() {
  const router = useRouter();
  const [formData, setFormData] = useState({
    title: '',
    description: '',
    price: '',
    category: '',
  });
  const [errors, setErrors] = useState({});
  const [loading, setLoading] = useState(false);

  async function handleSubmit(e) {
    e.preventDefault();
    setLoading(true);
    setErrors({});

    try {
      const res = await fetch('http://localhost:8000/api/listings', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('token')}`,
        },
        body: JSON.stringify(formData),
      });

      if (!res.ok) {
        const data = await res.json();
        if (res.status === 422 && data.errors) {
          setErrors(data.errors);
          return;
        }
        throw new Error(data.message || 'Failed to create listing');
      }

      const data = await res.json();
      router.push(`/listings/${data.id}`);
      router.refresh();
    } catch (error) {
      setErrors({ general: [error.message] });
    } finally {
      setLoading(false);
    }
  }

  return (
    <form onSubmit={handleSubmit} className="listing-form">
      {errors.general && (
        <div className="alert alert-error">{errors.general[0]}</div>
      )}

      <div className="form-group">
        <label htmlFor="title">Title</label>
        <input id="title" name="title" value={formData.title}
               onChange={(e) => setFormData(p => ({...p, title: e.target.value}))} />
        {errors.title && <p className="error">{errors.title[0]}</p>}
      </div>

      <div className="form-group">
        <label htmlFor="description">Description</label>
        <textarea id="description" name="description" value={formData.description}
                   onChange={(e) => setFormData(p => ({...p, description: e.target.value}))} />
        {errors.description && <p className="error">{errors.description[0]}</p>}
      </div>

      <div className="form-group">
        <label htmlFor="price">Price</label>
        <input id="price" name="price" type="number" value={formData.price}
               onChange={(e) => setFormData(p => ({...p, price: e.target.value}))} />
        {errors.price && <p className="error">{errors.price[0]}</p>}
      </div>

      <button type="submit" disabled={loading}>
        {loading ? 'Creating...' : 'Create Listing'}
      </button>
    </form>
  );
}

Quiz

Summary

  • Ang forms ay kailangang maging client components — gumagamit ng useState, event handlers, at browser interaction.
  • I-send ang form data gamit ang fetch with appropriate HTTP methods (POST, PUT, DELETE) at headers.
  • I-handle ang Laravel 422 validation errors — ang errors object ay may field names as keys at array of messages as values.
  • Gamitin ang router.refresh() pagkatapos ng mutation para i-update ang server-rendered data.
